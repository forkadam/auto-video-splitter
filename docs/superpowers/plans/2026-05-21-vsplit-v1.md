# vsplit v1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a local Python CLI that scans a folder of video files, detects scene boundaries, runs a human-review gate, and losslessly splits each file into clips — all resumable, failure-isolated, and Windows-first.

**Architecture:** Two layers (orchestrator + per-file pipeline) over filesystem-as-bus state. Seven stages per file: `probe → detect → preview → review → plan → split → verify`. Manifest JSON per workspace is the single source of truth for stage progress.

**Tech Stack:** Python 3.11+, `click`, `pydantic` v2, `pyyaml`, `scenedetect[opencv]`, `jinja2`, `rich`, `structlog`, `platformdirs`, FFmpeg/FFprobe ≥ 5.0.

**Reference spec:** `docs/superpowers/specs/2026-05-21-vsplit-design.md`

from __future__ import annotations
import re
from pathlib import Path

_RESERVED_NAMES = {
    "con", "prn", "aux", "nul",
    *(f"com{i}" for i in range(1, 10)),
    *(f"lpt{i}" for i in range(1, 10)),
}
_RESERVED_CHARS = re.compile(r'[<>:"/\\|?*]')
_WHITESPACE_UNDERSCORE = re.compile(r"[\s_]+")
_REPEATED_HYPHEN = re.compile(r"-{2,}")
_MAX_LEN = 64


def slugify(filename: str) -> str:
    stem = Path(filename).stem.lower()
    stem = _RESERVED_CHARS.sub("", stem)
    stem = _WHITESPACE_UNDERSCORE.sub("-", stem)
    stem = _REPEATED_HYPHEN.sub("-", stem)
    stem = stem.strip(". -")
    if stem in _RESERVED_NAMES:
        stem = f"_{stem}"
    return stem[:_MAX_LEN]


def resolve_collision(
    slug: str, live: set[str], retired: set[str]
) -> str:
    taken_lower = {s.lower() for s in live} | {s.lower() for s in retired}
    if slug.lower() not in taken_lower:
        return slug
    n = 2
    while f"{slug}-{n}".lower() in taken_lower:
        n += 1
    return f"{slug}-{n}"
```

from __future__ import annotations
import hashlib
from pathlib import Path

_CHUNK = 1024 * 1024  # 1 MB


def partial_hash(path: Path) -> str:
    """sha256(first 1MB + last 1MB + size_bytes_big_endian_8). Fast identity."""
    size = path.stat().st_size
    h = hashlib.sha256()
    with path.open("rb") as f:
        if size <= 2 * _CHUNK:
            h.update(f.read())
        else:
            h.update(f.read(_CHUNK))
            f.seek(-_CHUNK, 2)
            h.update(f.read(_CHUNK))
    h.update(size.to_bytes(8, "big"))
    return f"sha256:{h.hexdigest()}"


def content_hash(path: Path) -> str:
    """Full sha256 of file contents. Used for cuts.json change detection."""
    h = hashlib.sha256()
    with path.open("rb") as f:
        for chunk in iter(lambda: f.read(_CHUNK), b""):
            h.update(chunk)
    return f"sha256:{h.hexdigest()}"
```

from __future__ import annotations
import json
import os
import tempfile
from pathlib import Path
from typing import Any


def atomic_write_text(path: Path, content: str, *, encoding: str = "utf-8") -> None:
    """Write-then-rename. Survives crashes mid-write. Always LF line endings."""
    path.parent.mkdir(parents=True, exist_ok=True)
    fd, tmp = tempfile.mkstemp(prefix=".", dir=path.parent)
    try:
        with os.fdopen(fd, "w", encoding=encoding, newline="\n") as f:
            f.write(content)
        os.replace(tmp, path)
    except BaseException:
        try:
            os.unlink(tmp)
        except FileNotFoundError:
            pass
        raise


def atomic_write_json(path: Path, data: Any, *, indent: int = 2) -> None:
    atomic_write_text(path, json.dumps(data, indent=indent, default=str) + "\n")
```

        block.error = None

from __future__ import annotations
from pathlib import Path
import yaml
from vsplit.schemas.config import Config


def _read_yaml(path: Path) -> dict:
    raw = path.read_text(encoding="utf-8")
    data = yaml.safe_load(raw) or {}
    if not isinstance(data, dict):
        raise ValueError(f"config at {path} must be a YAML mapping")
    return data


def _deep_merge(base: dict, overlay: dict) -> dict:
    out = dict(base)
    for k, v in overlay.items():
        if isinstance(v, dict) and isinstance(out.get(k), dict):
            out[k] = _deep_merge(out[k], v)
        else:
            out[k] = v
    return out


def load_resolved_config(
    *, working_root: Path, explicit_path: Path | None
) -> Config:
    """Resolution order: defaults -> .vsplit/config.yaml -> explicit_path."""
    layers: list[dict] = [{}]   # defaults handled by pydantic
    candidate = working_root / ".vsplit" / "config.yaml"
    if candidate.exists():
        layers.append(_read_yaml(candidate))
    if explicit_path is not None and explicit_path.exists():
        layers.append(_read_yaml(explicit_path))
    merged: dict = {}
    for layer in layers:
        merged = _deep_merge(merged, layer)
    return Config.model_validate(merged)
```

from __future__ import annotations
from datetime import datetime, timezone
from pathlib import Path
import click
from rich.console import Console

from vsplit.core.config import load_resolved_config
from vsplit.core.hash import partial_hash
from vsplit.core.index import load_index, save_index
from vsplit.core.manifest import (
    load_manifest, save_manifest, new_manifest, manifest_path
)
from vsplit.core.paths import workspace_dir, working_state_dir
from vsplit.core.slugify import slugify, resolve_collision
from vsplit.schemas.index import WorkspaceEntry
from vsplit.schemas.manifest import SourceInfo

console = Console()


@click.command(name="scan")
@click.argument("path", required=False, type=click.Path(exists=True, file_okay=False))
@click.option("--recursive/--no-recursive", default=None)
@click.option("--dry-run", is_flag=True)
@click.option("--auto-relocate", is_flag=True, help="Update pointer without prompting on rename.")
@click.pass_context
def scan_cmd(ctx, path, recursive, dry_run, auto_relocate):
    """Discover videos and adopt new ones into workspaces."""
    working_root = Path(path) if path else Path(ctx.obj["working_dir"])
    config = load_resolved_config(
        working_root=working_root,
        explicit_path=ctx.obj.get("config_path"),
    )
    recursive_eff = config.scan.recursive if recursive is None else recursive
    extensions = {e.lower().lstrip(".") for e in config.scan.extensions}

    idx = load_index(working_root)
    live_slugs = set(idx.workspaces.keys())
    retired_slugs = set(idx.retired_slugs)
    by_hash = {w.partial_hash: slug for slug, w in idx.workspaces.items()}
    by_path = {w.source_relative_path: slug for slug, w in idx.workspaces.items()}

    iterator = (working_root.rglob("*") if recursive_eff else working_root.iterdir())
    candidates = []
    for p in iterator:
        if not p.is_file():
            continue
        if p.suffix.lower().lstrip(".") not in extensions:
            continue
        # Skip anything inside .vsplit
        try:
            rel = p.relative_to(working_root)
        except ValueError:
            continue
        if rel.parts and rel.parts[0] == ".vsplit":
            continue
        candidates.append(p)

    adopted = 0
    for p in sorted(candidates):
        rel = str(p.relative_to(working_root))
        stat = p.stat()
        ph = partial_hash(p)
        if ph in by_hash:
            existing_slug = by_hash[ph]
            entry = idx.workspaces[existing_slug]
            if entry.source_relative_path != rel:
                if not (auto_relocate or click.confirm(
                    f"Rename detected: {entry.source_relative_path} -> {rel}. Update pointer?",
                    default=False,
                )):
                    continue
                if not dry_run:
                    entry.source_relative_path = rel
                    ws = workspace_dir(working_root, existing_slug)
                    m = load_manifest(ws)
                    m.source.relative_path = rel
                    save_manifest(ws, m)
            continue
        if rel in by_path:
            console.print(
                f"[yellow]warning[/]: source {rel} replaced (hash changed); "
                f"existing workspace state may be stale"
            )
            continue
        base = slugify(p.name)
        if not base:
            console.print(f"[yellow]skip[/]: cannot slugify {rel}")
            continue
        new_slug = resolve_collision(base, live_slugs, retired_slugs)
        live_slugs.add(new_slug)
        if dry_run:
            console.print(f"[blue]would adopt[/]: {rel} -> {new_slug}")
            continue
        ws = workspace_dir(working_root, new_slug)
        ws.mkdir(parents=True, exist_ok=True)
        source = SourceInfo(
            relative_path=rel,
            size_bytes=stat.st_size,
            mtime=datetime.fromtimestamp(stat.st_mtime, tz=timezone.utc),
            partial_hash=ph,
        )
        m = new_manifest(slug=new_slug, source=source)
        save_manifest(ws, m)
        idx.workspaces[new_slug] = WorkspaceEntry(
            source_relative_path=rel,
            size_bytes=stat.st_size,
            partial_hash=ph,
            adopted_at=datetime.now(timezone.utc),
        )
        adopted += 1
        console.print(f"[green]adopted[/]: {rel} -> {new_slug}")

    if not dry_run:
        working_state_dir(working_root).mkdir(parents=True, exist_ok=True)
        save_index(working_root, idx)
    console.print(f"\n{adopted} adopted, {len(idx.workspaces)} total")
```

from rich.table import Table
from vsplit.schemas.manifest import STAGE_NAMES, Manifest

STAGE_HEADERS = {n: n.upper() for n in STAGE_NAMES}

_STATUS_GLYPH = {
    "done": "ok",
    "pending": "-",
    "running": "RUN",
    "failed": "FAIL",
    "awaiting_user": "wait",
    "skipped": "skip",
}


def render_status_table(manifests: dict[str, Manifest], console: Console) -> None:
    if not manifests:
        console.print("0 workspaces")
        return
    table = Table(show_header=True, header_style="bold")
    table.add_column("SLUG")
    for n in STAGE_NAMES:
        table.add_column(STAGE_HEADERS[n])
    for slug, m in sorted(manifests.items()):

from __future__ import annotations
import json
import subprocess
from vsplit.core.atomic import atomic_write_text, atomic_write_json
from vsplit.core.types import Stage, StageContext, StageResult


class ProbeStage(Stage):
    name = "probe"
    depends_on: list[str] = []
    output_subdir = "probe"

    def run(self, ctx: StageContext) -> StageResult:
        outdir = ctx.workspace_dir / "probe"
        outdir.mkdir(parents=True, exist_ok=True)
        src = str(ctx.source_path)

        meta_proc = subprocess.run(
            ["ffprobe", "-v", "error", "-print_format", "json",
             "-show_streams", "-show_format", src],
            check=True, capture_output=True, text=True,
        )
        meta = json.loads(meta_proc.stdout)
        atomic_write_json(outdir / "ffprobe.json", meta)

        kf_proc = subprocess.run(
            ["ffprobe", "-v", "error", "-select_streams", "v:0",
             "-skip_frame", "nokey", "-show_frames",
             "-show_entries", "frame=pts_time",
             "-of", "csv=p=0", src],
            check=True, capture_output=True, text=True,
        )
        atomic_write_text(outdir / "keyframes.csv", kf_proc.stdout)

        v_stream = next(
            (s for s in meta.get("streams", []) if s.get("codec_type") == "video"),
            None,
        )
        a_stream = next(
            (s for s in meta.get("streams", []) if s.get("codec_type") == "audio"),
            None,
        )
        fps = None
        if v_stream:
            num, _, den = v_stream.get("avg_frame_rate", "0/0").partition("/")
            try:
                if int(den) != 0:
                    fps = int(num) / int(den)
            except ValueError:
                fps = None

        keyframe_count = sum(1 for line in kf_proc.stdout.splitlines() if line.strip())

        ctx.manifest.metadata.duration_seconds = (
            float(meta.get("format", {}).get("duration", 0.0)) or None
        )
        ctx.manifest.metadata.container = meta.get("format", {}).get("format_name")
        ctx.manifest.metadata.video_codec = v_stream and v_stream.get("codec_name")
        ctx.manifest.metadata.audio_codec = a_stream and a_stream.get("codec_name")
        ctx.manifest.metadata.fps = fps
        ctx.manifest.metadata.width = v_stream and v_stream.get("width")
        ctx.manifest.metadata.height = v_stream and v_stream.get("height")
        ctx.manifest.metadata.keyframe_count = keyframe_count

        return StageResult(artifacts={"keyframe_count": keyframe_count})
```

```python
from __future__ import annotations
import logging
import threading
from datetime import datetime, timezone
from pathlib import Path
from vsplit.core.manifest import load_manifest, save_manifest
from vsplit.core.types import Stage, StageContext
from vsplit.schemas.config import Config
from vsplit.schemas.manifest import StageError


def _stage_logger(workspace_dir: Path, name: str) -> logging.Logger:
    logs = workspace_dir / "logs"
    logs.mkdir(parents=True, exist_ok=True)
    logger = logging.getLogger(f"vsplit.{workspace_dir.name}.{name}")
    logger.setLevel(logging.DEBUG)
    # Drop any prior handlers (test isolation)
    for h in list(logger.handlers):
        logger.removeHandler(h)
    handler = logging.FileHandler(logs / f"{name}.log", encoding="utf-8")
    handler.setFormatter(logging.Formatter("%(asctime)s %(levelname)s %(message)s"))
    logger.addHandler(handler)
    logger.propagate = False
    return logger


def run_stage(
    *,
    stage: Stage,
    workspace_dir: Path,
    working_root: Path,
    source_path: Path,
    config: Config,
    cancel_event: threading.Event,
) -> None:
    """Run one stage, persist status before and after."""
    manifest = load_manifest(workspace_dir)
    block = manifest.stages[stage.name]
    block.status = "running"
    block.started_at = datetime.now(timezone.utc)
    block.error = None
    save_manifest(workspace_dir, manifest)

    logger = _stage_logger(workspace_dir, stage.name)
    ctx = StageContext(
        workspace_dir=workspace_dir,
        working_root=working_root,
        source_path=source_path,
        manifest=manifest,
        config=config,
        logger=logger,
        cancel_event=cancel_event,
    )

    try:
        result = stage.run(ctx)
    except Exception as exc:
        logger.exception("stage failed")
        block.status = "failed"
        block.error = StageError(
            message=str(exc),
            exit_code=getattr(exc, "returncode", None),
            log_path=str((workspace_dir / "logs" / f"{stage.name}.log").relative_to(working_root)),
        )
        save_manifest(workspace_dir, manifest)
        return

    block.status = result.status
    if result.status == "awaiting_user":
        block.set_at = datetime.now(timezone.utc)
    else:
        block.completed_at = datetime.now(timezone.utc)
    if result.artifacts:
        block.artifacts.update(result.artifacts)
    save_manifest(workspace_dir, manifest)
```

from __future__ import annotations
import threading
from pathlib import Path
import click
from rich.console import Console
from vsplit.core.config import load_resolved_config
from vsplit.core.index import load_index
from vsplit.core.manifest import load_manifest
from vsplit.core.paths import workspace_dir
from vsplit.core.pipeline import run_stage
from vsplit.schemas.manifest import STAGE_NAMES

console = Console()

# Registry will be populated as stages are added; for now, just probe.
def _stage_instance(name: str):
    from vsplit.stages.probe import ProbeStage
    REGISTRY = {"probe": ProbeStage}
    if name not in REGISTRY:
        raise click.ClickException(f"stage '{name}' not implemented yet")
    return REGISTRY[name]()


@click.command(name="run")
@click.argument("slug_glob", required=False)
@click.option("--only", "only_stage", default=None,
              type=click.Choice(list(STAGE_NAMES)))
@click.option("--from", "from_stage", default=None,
              type=click.Choice(list(STAGE_NAMES)))
@click.option("--through", "through_stage", default=None,
              type=click.Choice(list(STAGE_NAMES)))
@click.option("--filter", "filters", multiple=True)
@click.option("--dry-run", is_flag=True)
@click.pass_context
def run_cmd(ctx, slug_glob, only_stage, from_stage, through_stage, filters, dry_run):
    """Advance workspaces through the pipeline."""
    working_root = Path(ctx.obj["working_dir"])
    config = load_resolved_config(working_root=working_root,
                                  explicit_path=ctx.obj.get("config_path"))
    idx = load_index(working_root)

    # Determine which stages this invocation will run.
    if only_stage:
        stages_to_run = [only_stage]
    else:
        start = STAGE_NAMES.index(from_stage) if from_stage else 0
        end = STAGE_NAMES.index(through_stage) + 1 if through_stage else len(STAGE_NAMES)
        stages_to_run = list(STAGE_NAMES[start:end])

    # Filter workspaces by slug glob (other filters added in later tasks).
    import fnmatch
    slugs = list(idx.workspaces.keys())
    if slug_glob:
        slugs = [s for s in slugs if fnmatch.fnmatch(s, slug_glob)]

    cancel_event = threading.Event()
    for slug in slugs:
        ws = workspace_dir(working_root, slug)
        manifest = load_manifest(ws)
        source = working_root / manifest.source.relative_path
        for stage_name in stages_to_run:
            block = manifest.stages[stage_name]
            if block.status == "done":
                continue
            if block.status in ("failed", "awaiting_user"):
                console.print(f"[yellow]{slug}/{stage_name}[/]: blocked ({block.status})")
                break
            try:
                stage = _stage_instance(stage_name)
            except click.ClickException as e:
                console.print(f"[yellow]{e}[/]")
                break
            if dry_run:
                console.print(f"[blue]would run[/]: {slug}/{stage_name}")
                continue
            console.print(f"[green]{slug}/{stage_name}[/]: running")
            run_stage(
                stage=stage,
                workspace_dir=ws,
                working_root=working_root,
                source_path=source,
                config=config,
                cancel_event=cancel_event,
            )
            manifest = load_manifest(ws)
            if manifest.stages[stage_name].status != "done":
                break
```

from scenedetect import detect, AdaptiveDetector
from vsplit.core.atomic import atomic_write_json
from vsplit.core.types import Stage, StageContext, StageResult


class DetectStage(Stage):
    name = "detect"
    depends_on = ["probe"]
    output_subdir = "detect"

    def run(self, ctx: StageContext) -> StageResult:
        outdir = ctx.workspace_dir / "detect"
        outdir.mkdir(parents=True, exist_ok=True)
        detector = AdaptiveDetector(
            adaptive_threshold=ctx.config.detect.threshold,
            min_scene_len=int(
                ctx.config.detect.min_scene_length * (ctx.manifest.metadata.fps or 24)
            ) or 1,
            luma_only=ctx.config.detect.luma_only,
        )
        scene_list = detect(str(ctx.source_path), detector)
        scenes_payload = []
        for i, (start, end) in enumerate(scene_list, start=1):
            scenes_payload.append({
                "id": i,
                "start_time": start.get_seconds(),
                "end_time": end.get_seconds(),
                "start_frame": start.get_frames(),
                "end_frame": end.get_frames(),
            })
        atomic_write_json(outdir / "scenes.json", {
            "detector": "adaptive",
            "config": {"threshold": ctx.config.detect.threshold},
            "scenes": scenes_payload,
        })
        return StageResult(artifacts={
            "scenes_found": len(scenes_payload),
            "detector": "adaptive",
        })

from __future__ import annotations
import json
import subprocess
from pathlib import Path
from jinja2 import Environment, FileSystemLoader, select_autoescape
from vsplit.core.atomic import atomic_write_text
from vsplit.core.types import Stage, StageContext, StageResult

_TEMPLATE_DIR = Path(__file__).parent.parent / "templates"


def _format_tc(seconds: float) -> str:
    h = int(seconds // 3600)
    m = int((seconds % 3600) // 60)
    s = seconds - h * 3600 - m * 60
    return f"{h:02d}:{m:02d}:{s:06.3f}"


class PreviewStage(Stage):
    name = "preview"
    depends_on = ["detect"]
    output_subdir = "preview"

    def run(self, ctx: StageContext) -> StageResult:
        outdir = ctx.workspace_dir / "preview"
        imgs = outdir / "images"
        imgs.mkdir(parents=True, exist_ok=True)

        scenes_data = json.loads(
            (ctx.workspace_dir / "detect" / "scenes.json").read_text()
        )
        n = ctx.config.preview.images_per_scene
        w = ctx.config.preview.image_width
        q = ctx.config.preview.image_quality

        scenes_for_template = []
        for scene in scenes_data["scenes"]:
            duration = scene["end_time"] - scene["start_time"]
            sample_ts = [
                scene["start_time"] + duration * (i / max(n - 1, 1))
                for i in range(n)
            ] if n > 1 else [scene["start_time"] + duration / 2]
            img_names = []
            for k, t in enumerate(sample_ts, start=1):
                name = f"scene-{scene['id']:03d}-{k:02d}.jpg"
                subprocess.run(
                    ["ffmpeg", "-y", "-loglevel", "error", "-ss", f"{t:.3f}",
                     "-i", str(ctx.source_path), "-frames:v", "1",
                     "-vf", f"scale={w}:-1",
                     "-q:v", str(max(2, min(31, 31 - int(q / 100 * 29)))),
                     str(imgs / name)],
                    check=True,
                )
                img_names.append(name)
            scenes_for_template.append({
                "id": scene["id"],
                "start_tc": _format_tc(scene["start_time"]),
                "end_tc": _format_tc(scene["end_time"]),
                "duration": duration,
                "images": img_names,
            })

        env = Environment(
            loader=FileSystemLoader(str(_TEMPLATE_DIR)),
            autoescape=select_autoescape(["html"]),
        )
        tmpl = env.get_template("preview_index.html.j2")
        rendered = tmpl.render(
            slug=ctx.manifest.id,
            source_rel=ctx.manifest.source.relative_path,
            scenes=scenes_for_template,
        )
        atomic_write_text(outdir / "index.html", rendered)

        return StageResult(artifacts={
            "images_generated": sum(len(s["images"]) for s in scenes_for_template),
        })

from __future__ import annotations
import json
from pydantic import ValidationError
from vsplit.core.atomic import atomic_write_text
from vsplit.core.hash import content_hash
from vsplit.core.types import Stage, StageContext, StageResult
from vsplit.schemas.cuts import CutsJson


def _format_tc(seconds: float) -> str:
    h = int(seconds // 3600)
    m = int((seconds % 3600) // 60)
    s = seconds - h * 3600 - m * 60
    return f"{h:02d}:{m:02d}:{s:06.3f}"


def _initial_cuts(scenes: list[dict], default_keep: bool) -> dict:
    return {
        "schema_version": 1,
        "scenes": [
            {
                "id": s["id"],
                "start": _format_tc(s["start_time"]),
                "end": _format_tc(s["end_time"]),
                "keep": default_keep,
                "label": None,
                "merge_with_next": False,
            }
            for s in scenes
        ],
    }


class ReviewStage(Stage):
    name = "review"
    depends_on = ["preview"]
    output_subdir = "decisions"

    def run(self, ctx: StageContext) -> StageResult:
        decisions_dir = ctx.workspace_dir / "decisions"
        decisions_dir.mkdir(parents=True, exist_ok=True)
        cuts_path = decisions_dir / "cuts.json"

        scenes_data = json.loads(
            (ctx.workspace_dir / "detect" / "scenes.json").read_text()
        )
        detected_ids = {s["id"] for s in scenes_data["scenes"]}

        block = ctx.manifest.stages["review"]
        is_first_entry = not cuts_path.exists()
        template_path = ctx.working_root / "cuts.json.template"

        # Template auto-apply (only on first entry)
        if (
            is_first_entry
            and ctx.config.review.auto_apply_template
            and template_path.exists()
        ):
            atomic_write_text(cuts_path, template_path.read_text() + ("" if template_path.read_text().endswith("\n") else "\n"))
            cuts = CutsJson.model_validate_json(cuts_path.read_text())
            _validate_ids(cuts, detected_ids)
            return StageResult(
                status="done",
                artifacts={
                    "cuts_file_hash": content_hash(cuts_path),
                    "scenes_total": len(cuts.scenes),
                    "scenes_marked_keep_final": sum(1 for s in cuts.scenes if s.keep),
                    "applied_template": True,
                },
            )

        if is_first_entry:
            cuts = _initial_cuts(scenes_data["scenes"], ctx.config.review.default_keep)
            atomic_write_text(cuts_path, json.dumps(cuts, indent=2) + "\n")
            return StageResult(
                status="awaiting_user",
                artifacts={
                    "cuts_file_hash": content_hash(cuts_path),
                    "scenes_total": len(cuts["scenes"]),
                    "scenes_marked_keep_initial": sum(1 for s in cuts["scenes"] if s["keep"]),
                },
            )

        # Re-entry: decide if the gate has been passed
        current_hash = content_hash(cuts_path)
        recorded_hash = block.artifacts.get("cuts_file_hash")
        approved = block.artifacts.get("approved_at") is not None

        if current_hash == recorded_hash and not approved:
            return StageResult(
                status="awaiting_user",
                artifacts={**block.artifacts},
            )

        try:
            parsed = CutsJson.model_validate_json(cuts_path.read_text())
        except ValidationError as e:
            raise ValueError(f"cuts.json invalid: {e}") from e

        _validate_ids(parsed, detected_ids)

        return StageResult(
            status="done",
            artifacts={
                **block.artifacts,
                "cuts_file_hash": current_hash,
                "scenes_marked_keep_final": sum(1 for s in parsed.scenes if s.keep),
            },
        )


def _validate_ids(parsed: CutsJson, detected_ids: set[int]) -> None:
    cuts_ids = {s.id for s in parsed.scenes}
    missing = detected_ids - cuts_ids
    extra = cuts_ids - detected_ids
    if missing:
        raise ValueError(
            f"scene IDs {sorted(missing)} from detect/scenes.json missing in cuts.json — "
            f"use keep:false to drop a scene, don't remove the row"
        )
    if extra:
        raise ValueError(
            f"scene IDs {sorted(extra)} in cuts.json not present in detect/scenes.json — "
            f"add new scenes by editing detect/scenes.json then `reset --from detect`"
        )

from __future__ import annotations
import json
from vsplit.core.atomic import atomic_write_json
from vsplit.core.types import Stage, StageContext, StageResult
from vsplit.schemas.cuts import CutsJson, Scene


def _parse_tc(tc: str) -> float:
    h, m, s = tc.split(":")
    return int(h) * 3600 + int(m) * 60 + float(s)


def _snap_preceding(t: float, keyframes: list[float]) -> float:
    candidates = [k for k in keyframes if k <= t]
    return candidates[-1] if candidates else 0.0


def _sanitize_label(label: str | None) -> str | None:
    if not label:
        return None
    safe = "".join(c if c.isalnum() or c in "-_" else "-" for c in label.lower())
    safe = "-".join(filter(None, safe.split("-")))
    return safe or None


class PlanStage(Stage):
    name = "plan"
    depends_on = ["review", "probe"]
    output_subdir = "plan"

    def run(self, ctx: StageContext) -> StageResult:
        outdir = ctx.workspace_dir / "plan"
        outdir.mkdir(parents=True, exist_ok=True)

        cuts = CutsJson.model_validate_json(
            (ctx.workspace_dir / "decisions" / "cuts.json").read_text()
        )
        keyframes = [
            float(line) for line in
            (ctx.workspace_dir / "probe" / "keyframes.csv").read_text().splitlines()
            if line.strip()
        ]
        kept = [s for s in cuts.scenes if s.keep]

        # Merge adjacent scenes via merge_with_next
        merged: list[list[Scene]] = []
        for s in kept:
            if merged and merged[-1][-1].merge_with_next:
                merged[-1].append(s)
            else:
                merged.append([s])

        resolved = []
        out_index = 1
        for group in merged:
            start_req = _parse_tc(group[0].start)
            end_req = _parse_tc(group[-1].end)
            if end_req - start_req < ctx.config.plan.min_segment_length:
                continue
            actual_start = _snap_preceding(start_req, keyframes)
            actual_end = end_req
            label = _sanitize_label(group[0].label)
            base = f"{ctx.manifest.id}-highlight-{out_index:02d}"
            fname = f"{base}-{label}.{ctx.config.split.output_container}" if label \
                else f"{base}.{ctx.config.split.output_container}"
            resolved.append({
                "id": group[0].id,
                "label": label,
                "requested_start": start_req,
                "requested_end": end_req,
                "actual_start": actual_start,
                "actual_end": actual_end,
                "drift_seconds": round(start_req - actual_start, 6),
                "strategy": "stream_copy",
                "output_filename": fname,
            })
            out_index += 1

        atomic_write_json(outdir / "cuts-resolved.json", {"scenes": resolved})
        return StageResult(artifacts={"segments": len(resolved)})

from __future__ import annotations
import json
import subprocess
from vsplit.core.types import Stage, StageContext, StageResult


class SplitStage(Stage):
    name = "split"
    depends_on = ["plan"]
    output_subdir = "output"

    def run(self, ctx: StageContext) -> StageResult:
        outdir = ctx.workspace_dir / "output"
        outdir.mkdir(parents=True, exist_ok=True)
        plan = json.loads(
            (ctx.workspace_dir / "plan" / "cuts-resolved.json").read_text()
        )
        produced = []
        for seg in plan["scenes"]:
            if ctx.cancel_event.is_set():
                break
            out_path = outdir / seg["output_filename"]
            cmd = [
                "ffmpeg", "-y", "-loglevel", "error",
                "-ss", f"{seg['actual_start']:.3f}",
                "-to", f"{seg['actual_end']:.3f}",
                "-i", str(ctx.source_path),
                "-map", "0", "-c", "copy",
                "-avoid_negative_ts", "make_zero",
                str(out_path),
            ]
            ctx.logger.info("split: %s", " ".join(cmd))
            subprocess.run(cmd, check=True)
            produced.append(seg["output_filename"])
        return StageResult(artifacts={"outputs": produced})

    def recover(self, ctx: StageContext) -> None:
        """Keep validated outputs; delete incomplete ones; reset to pending."""
        outdir = ctx.workspace_dir / "output"
        plan_path = ctx.workspace_dir / "plan" / "cuts-resolved.json"
        if plan_path.exists() and outdir.exists():
            plan = json.loads(plan_path.read_text())
            expected = {s["output_filename"] for s in plan["scenes"]}
            for f in list(outdir.iterdir()):
                if f.name not in expected:
                    f.unlink(missing_ok=True)
                    continue
                try:
                    subprocess.run(
                        ["ffprobe", "-v", "error", "-show_format", str(f)],
                        check=True, capture_output=True,
                    )
                except subprocess.CalledProcessError:
                    f.unlink(missing_ok=True)
        block = ctx.manifest.stages[self.name]
        block.status = "pending"
        block.artifacts = {}
        block.error = None
        block.started_at = None
        block.completed_at = None

from __future__ import annotations
import fnmatch
from dataclasses import dataclass
from typing import Callable


@dataclass(frozen=True)
class Predicate:
    key: str
    value: str


VALID_KEYS = {"status", "stage", "slug"}


def parse_filter(s: str) -> Predicate:
    if "=" not in s:
        raise ValueError(f"filter must be key=value, got {s!r}")
    k, v = s.split("=", 1)
    if k not in VALID_KEYS:
        raise ValueError(f"unknown filter key: {k!r} (allowed: {sorted(VALID_KEYS)})")
    return Predicate(key=k, value=v)


def apply_filters(
    slugs: list[str],
    predicates: list[Predicate],
    *,
    status_pred: Callable[[str, str], bool],
    stage_pred: Callable[[str, str], bool],
) -> list[str]:
    result = list(slugs)
    for p in predicates:
        if p.key == "slug":
            result = [s for s in result if fnmatch.fnmatch(s, p.value)]
        elif p.key == "status":
            result = [s for s in result if status_pred(s, p.value)]
        elif p.key == "stage":
            result = [s for s in result if stage_pred(s, p.value)]
    return result
```

```python
from __future__ import annotations
import threading
from concurrent.futures import ThreadPoolExecutor, Future, wait, FIRST_COMPLETED
from dataclasses import dataclass
from pathlib import Path
from typing import Callable
from vsplit.core.manifest import load_manifest
from vsplit.core.pipeline import run_stage
from vsplit.core.types import Stage
from vsplit.schemas.config import Config
from vsplit.schemas.manifest import STAGE_NAMES


@dataclass(frozen=True)
class Workspace:
    slug: str
    workspace_dir: Path
    source_path: Path


def _next_runnable(manifest, stages_to_run: list[str]) -> str | None:
    for stage in stages_to_run:
        block = manifest.stages[stage]
        if block.status == "done":
            continue
        if block.status in ("failed", "awaiting_user", "running"):
            return None
        if block.status == "pending":
            # All earlier stages must be done.
            earlier = STAGE_NAMES[: STAGE_NAMES.index(stage)]
            if any(manifest.stages[e].status != "done" for e in earlier):
                return None
            return stage
        if block.status == "skipped":
            continue
    return None


def _recover_running(
    workspaces: list[Workspace],
    working_root: Path,
    config: Config,
    cancel_event: threading.Event,
    stage_factory: Callable[[str], Stage],
) -> None:
    """Before dispatch, walk each manifest and call recover() on any running stage."""
    import logging
    for ws in workspaces:
        manifest = load_manifest(ws.workspace_dir)
        for name, block in manifest.stages.items():
            if block.status != "running":
                continue
            stage = stage_factory(name)
            from vsplit.core.types import StageContext
            ctx = StageContext(
                workspace_dir=ws.workspace_dir,
                working_root=working_root,
                source_path=ws.source_path,
                manifest=manifest,
                config=config,
                logger=logging.getLogger(f"vsplit.recover.{ws.slug}.{name}"),
                cancel_event=cancel_event,
            )
            stage.recover(ctx)
        from vsplit.core.manifest import save_manifest
        save_manifest(ws.workspace_dir, manifest)


def dispatch(
    *,
    workspaces: list[Workspace],
    working_root: Path,
    config: Config,
    cancel_event: threading.Event,
    stages_to_run: list[str],
    stage_factory: Callable[[str], Stage],
) -> None:
    _recover_running(workspaces, working_root, config, cancel_event, stage_factory)

    limiters = {s: threading.Semaphore(config.concurrency.get(s, 1))
                for s in stages_to_run}
    inflight: dict[str, tuple[Future, str]] = {}
    max_workers = max(config.concurrency.get(s, 1) for s in stages_to_run) or 1

    with ThreadPoolExecutor(max_workers=max_workers * len(stages_to_run)) as ex:
        while not cancel_event.is_set():
            # Reap
            done_slugs = []
            for slug, (fut, stage_name) in list(inflight.items()):
                if fut.done():
                    try:
                        fut.result()
                    except Exception:
                        pass    # error already in manifest
                    limiters[stage_name].release()
                    done_slugs.append(slug)
            for s in done_slugs:
                del inflight[s]

            # Schedule
            scheduled = False
            for ws in workspaces:
                if ws.slug in inflight:
                    continue
                manifest = load_manifest(ws.workspace_dir)
                nxt = _next_runnable(manifest, stages_to_run)
                if nxt is None:
                    continue
                limiter = limiters[nxt]
                if not limiter.acquire(blocking=False):
                    continue
                stage = stage_factory(nxt)
                fut = ex.submit(
                    run_stage,
                    stage=stage,
                    workspace_dir=ws.workspace_dir,
                    working_root=working_root,
                    source_path=ws.source_path,
                    config=config,
                    cancel_event=cancel_event,
                )
                inflight[ws.slug] = (fut, nxt)
                scheduled = True

            if not inflight and not scheduled:
                return
            if inflight:
                wait([f for f, _ in inflight.values()],
                     return_when=FIRST_COMPLETED, timeout=0.5)
```

def _stage_instance(name: str):
    from vsplit.stages.probe import ProbeStage
    from vsplit.stages.detect import DetectStage
    from vsplit.stages.preview import PreviewStage
    from vsplit.stages.review import ReviewStage
    from vsplit.stages.plan import PlanStage
    from vsplit.stages.split import SplitStage
    from vsplit.stages.verify import VerifyStage
    REGISTRY = {
        "probe": ProbeStage, "detect": DetectStage,
        "preview": PreviewStage, "review": ReviewStage,
        "plan": PlanStage, "split": SplitStage, "verify": VerifyStage,
    }
    return REGISTRY[name]()


@click.command(name="run")
@click.argument("slug_glob", required=False)
@click.option("--only", "only_stage", default=None,
              type=click.Choice(list(STAGE_NAMES)))
@click.option("--from", "from_stage", default=None,
              type=click.Choice(list(STAGE_NAMES)))
@click.option("--through", "through_stage", default=None,
              type=click.Choice(list(STAGE_NAMES)))
@click.option("--filter", "filters", multiple=True)
@click.option("--concurrency", "concurrency_overrides", multiple=True,
              help="stage=N, e.g. split=4")
@click.option("--dry-run", is_flag=True)
@click.pass_context
def run_cmd(ctx, slug_glob, only_stage, from_stage, through_stage,
            filters, concurrency_overrides, dry_run):
    """Advance workspaces through the pipeline."""
    working_root = Path(ctx.obj["working_dir"])
    config = load_resolved_config(working_root=working_root,
                                  explicit_path=ctx.obj.get("config_path"))

    for override in concurrency_overrides:
        if "=" not in override:
            raise click.ClickException(f"bad --concurrency: {override}")
        k, v = override.split("=", 1)
        config.concurrency[k] = int(v)

    if only_stage:
        stages_to_run = [only_stage]
    else:
        start = STAGE_NAMES.index(from_stage) if from_stage else 0
        end = STAGE_NAMES.index(through_stage) + 1 if through_stage else len(STAGE_NAMES)
        stages_to_run = list(STAGE_NAMES[start:end])

    idx = load_index(working_root)
    slugs = list(idx.workspaces.keys())
    preds: list[Predicate] = []
    if slug_glob:
        preds.append(Predicate(key="slug", value=slug_glob))
    for f in filters:
        preds.append(parse_filter(f))

    def has_status(slug: str, val: str) -> bool:
        m = load_manifest(workspace_dir(working_root, slug))
        return any(b.status == val for b in m.stages.values())

    def at_stage(slug: str, val: str) -> bool:
        m = load_manifest(workspace_dir(working_root, slug))
        # workspace is "at stage X" if X is the next non-done stage
        for s in STAGE_NAMES:
            if m.stages[s].status != "done":
                return s == val
        return False

    slugs = apply_filters(slugs, preds, status_pred=has_status, stage_pred=at_stage)

    workspaces = []
    for s in slugs:
        ws = workspace_dir(working_root, s)
        m = load_manifest(ws)
        workspaces.append(Workspace(
            slug=s,
            workspace_dir=ws,
            source_path=working_root / m.source.relative_path,
        ))

    if dry_run:
        for w in workspaces:
            console.print(f"[blue]would run[/]: {w.slug} stages={stages_to_run}")
        return

    cancel_event = threading.Event()

    def _on_sigint(signum, frame):
        console.print("\n[yellow]cancellation requested; finishing in-flight stages...[/]")
        cancel_event.set()

    signal.signal(signal.SIGINT, _on_sigint)

    dispatch(
        workspaces=workspaces,
        working_root=working_root,
        config=config,
        cancel_event=cancel_event,
        stages_to_run=stages_to_run,
        stage_factory=_stage_instance,
    )

    if cancel_event.is_set():
        ctx.exit(130)

from vsplit.core.slugify import slugify
from vsplit.schemas.manifest import STAGE_NAMES

console = Console()


@click.command(name="one")
@click.argument("file_path", type=click.Path(exists=True, dir_okay=False))
@click.option("--through", "through_stage", default=None,
              type=click.Choice(list(STAGE_NAMES)))
@click.option("--auto-approve", is_flag=True)
@click.pass_context
def one_cmd(ctx, file_path, through_stage, auto_approve):
    """Process a single file end-to-end. Implies scan + run."""
    file_path = Path(file_path).resolve()
    parent = file_path.parent
    # rewrite working dir to the parent of the file
    ctx.obj["working_dir"] = parent

    # 1. scan parent (idempotent; picks up the file)
    ctx.invoke(scan_cmd)

    slug = slugify(file_path.name)

    # 2. run through preview (or through_stage if given)
    target_through = through_stage or "preview"
    ctx.invoke(run_cmd, slug_glob=slug, through_stage=target_through)

    # 3. open preview HTML if it exists
    preview_html = workspace_dir(parent, slug) / "preview" / "index.html"
    if preview_html.exists():
        webbrowser.open(preview_html.as_uri())
        console.print(f"opened {preview_html}")

    if auto_approve:
        # finish review (will fall through to plan/split/verify on next run)
        ctx.invoke(run_cmd, slug_glob=slug, through_stage="review")
        ctx.invoke(approve_cmd, slug_glob=slug)
        ctx.invoke(run_cmd, slug_glob=slug)
```

    try:
        r = subprocess.run(cmd, capture_output=True, text=True, check=True)
        return r.stdout.splitlines()[0].strip() if r.stdout else None
    except (FileNotFoundError, subprocess.CalledProcessError, IndexError):

# vsplit

Local, scene-aware video splitter with a human review gate. Built for batches of personal video.

## Install

Prerequisites: Python 3.11+, FFmpeg, FFprobe.

pip install -e ".[dev]"

## Quick start

```bash
cd ~/vacation-clips
vsplit scan                     # adopt every video as a workspace
vsplit run --through preview    # detect scenes, generate thumbnails
open .vsplit/workspaces/*/preview/index.html
# edit .vsplit/workspaces/*/decisions/cuts.json (set keep:false on scenes you don't want)
vsplit run                      # plan, split, verify
ls .vsplit/workspaces/*/output/ # your clips
```

## Single-file shortcut

```bash
vsplit one Birthday.mov              # scan + run + open preview
vsplit one Birthday.mov --auto-approve   # skip the review gate
```

## Commands

| Command | Purpose |
|--|--|
| `vsplit scan [path]` | Discover and adopt video files |
| `vsplit run [filter]` | Advance pipelines |
| `vsplit status` | Show pipeline progress |
| `vsplit approve <filter>` | Mark workspaces past the review gate |
| `vsplit reset <filter> --from <stage>` | Re-run stages |
| `vsplit clean <filter>` | Delete workspace state (source files untouched) |
| `vsplit one <file>` | Single-file shortcut |

## Filter grammar

`--filter key=value` where key is `status`, `stage`, or `slug`. Repeat to AND. Bare positional argument is a slug glob.

```
vsplit run --filter status=awaiting_user
vsplit run "vacation-*"
vsplit clean --filter status=done
```

## Config

`.vsplit/config.yaml` overrides defaults. See `docs/superpowers/specs/2026-05-21-vsplit-design.md` §3.4 for the full schema.

## Windows notes

- Run from a path short enough that workspace files stay under MAX_PATH. The tool warns at scan time if you're close.
- Exclude your working folder from real-time antivirus scanning during heavy operations.
